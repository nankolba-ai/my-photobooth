<script>
  // 📌 นำ Web App URL ที่ได้จาก Step 1 มาวางที่นี่
  const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbw5lhzcxE28syFgbaqW2u3j0Lk776vMzKkaNDT48-5xb6z0k7Bmdfuut5YwWVELwHf4/exec"; 

  const video = document.getElementById('webcam');
  const cameraSelect = document.getElementById('cameraSelect');
  const countdownEl = document.getElementById('countdown');
  const resultModal = document.getElementById('resultModal');
  const finalPhoto = document.getElementById('finalPhoto');
  const qrcodeDiv = document.getElementById('qrcode');
  const loader = document.getElementById('loader');

  let photos = [];
  const TOTAL_SHOTS = 3; // จำนวนช็อตที่ถ่าย (3 รูป)

  // 1. ดึงรายการกล้องที่เชื่อมต่อทั้งหมด
  async function getCameras() {
    const devices = await navigator.mediaDevices.enumerateDevices();
    const videoDevices = devices.filter(device => device.kind === 'videoinput');
    cameraSelect.innerHTML = '';
    videoDevices.forEach((device, index) => {
      const option = document.createElement('option');
      option.value = device.deviceId;
      option.text = device.label || `กล้อง ${index + 1}`;
      cameraSelect.appendChild(option);
    });
  }

  // 2. เปิดใช้งานกล้องที่เลือก
  async function startCamera(deviceId) {
    const constraints = {
      video: {
        deviceId: deviceId ? { exact: deviceId } : undefined,
        width: { ideal: 1920 },
        height: { ideal: 1080 }
      }
    };
    const stream = await navigator.mediaDevices.getUserMedia(constraints);
    video.srcObject = stream;
  }

  cameraSelect.addEventListener('change', () => startCamera(cameraSelect.value));

  // 3. เริ่มขั้นตอนนับถอยหลังถ่ายรูป
  async function startPhotoBooth() {
    document.getElementById('startBtn').disabled = true;
    photos = [];

    for (let i = 0; i < TOTAL_SHOTS; i++) {
      await runCountdown(3); // นับถอยหลัง 3 วินาทีต่อภาพ
      takeSnapshot();
      await new Promise(r => setTimeout(r, 1000)); // พัก 1 วินาทีระหว่างช็อต
    }

    generatePhotoStrip();
    document.getElementById('startBtn').disabled = false;
  }

  function runCountdown(seconds) {
    return new Promise((resolve) => {
      countdownEl.style.display = 'block';
      let count = seconds;
      countdownEl.innerText = count;

      let timer = setInterval(() => {
        count--;
        if (count > 0) {
          countdownEl.innerText = count;
        } else {
          clearInterval(timer);
          countdownEl.style.display = 'none';
          resolve();
        }
      }, 1000);
    });
  }

  // 4. บันทึกภาพแต่ละช็อตลง Memory
  function takeSnapshot() {
    const canvas = document.createElement('canvas');
    canvas.width = video.videoWidth;
    canvas.height = video.videoHeight;
    const ctx = canvas.getContext('2d');
    
    // กลับด้านภาพ (Mirroring) ให้ตรงกับมุมมอง
    ctx.translate(canvas.width, 0);
    ctx.scale(-1, 1);
    ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
    
    photos.push(canvas);
  }

  // 5. นำภาพมารวมกันเป็น Photo Strip (กรอบ Photo Booth)
  function generatePhotoStrip() {
    const canvas = document.getElementById('hiddenCanvas');
    const ctx = canvas.getContext('2d');

    const photoWidth = 600;
    const photoHeight = 400;
    const padding = 30;
    const headerSpace = 40;
    const footerSpace = 100;

    canvas.width = photoWidth + (padding * 2);
    canvas.height = (photoHeight * TOTAL_SHOTS) + (padding * (TOTAL_SHOTS + 1)) + footerSpace;

    // พื้นหลังแถบภาพ (สีขาว/มินิมอล)
    ctx.fillStyle = '#ffffff';
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    // วาดภาพแต่ละช็อตลงบน Frame (ปรับเป็นโทน B&W หากต้องการ)
    photos.forEach((imgCanvas, index) => {
      const yPos = padding + index * (photoHeight + padding);
      
      // ปรับภาพให้เป็นขาวดำ (สไตล์ภาพตัวอย่าง)
      ctx.filter = 'grayscale(100%) contrast(110%)';
      ctx.drawImage(imgCanvas, padding, yPos, photoWidth, photoHeight);
      ctx.filter = 'none'; // รีเซ็ตฟิลเตอร์
    });

    // เพิ่มข้อความด้านล่าง (Footer Branding)
    ctx.fillStyle = '#000000';
    ctx.font = 'italic 20px Georgia';
    ctx.textAlign = 'center';
    const today = new Date().toLocaleDateString('th-TH');
    ctx.fillText(`MEMORIES • ${today}`, canvas.width / 2, canvas.height - 40);

    const dataUrl = canvas.toDataURL('image/png');
    finalPhoto.src = dataUrl;
    
    // แสดง Modal และอัปโหลดไปยัง Google Drive
    resultModal.style.display = 'flex';
    uploadToGoogleDrive(dataUrl);
  }

  // 6. อัปโหลดภาพไปยัง Google Apps Script
  function uploadToGoogleDrive(base64Image) {
    qrcodeDiv.innerHTML = '';
    loader.style.display = 'block';

    fetch(GOOGLE_SCRIPT_URL, {
      method: 'POST',
      body: JSON.stringify({ image: base64Image })
    })
    .then(res => res.json())
    .then(data => {
      loader.style.display = 'none';
      if (data.status === 'success') {
        // สร้าง QR Code จาก URL สำหรับดาวน์โหลดภาพ
        new QRCode(qrcodeDiv, {
          text: data.downloadUrl,
          width: 150,
          height: 150
        });
      } else {
        alert('เกิดข้อผิดพลาดในการอัปโหลดภาพ');
      }
    })
    .catch(err => {
      loader.style.display = 'none';
      console.error(err);
    });
  }

  function closeModal() {
    resultModal.style.display = 'none';
  }

  // เริ่มต้นระบบ
  navigator.mediaDevices.getUserMedia({ video: true }).then(() => {
    getCameras().then(() => {
      if (cameraSelect.options.length > 0) {
        startCamera(cameraSelect.value);
      }
    });
  });
</script>
